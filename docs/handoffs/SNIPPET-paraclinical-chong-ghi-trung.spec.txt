// Cách dùng: chèn NGUYÊN khối bên dưới vào apps/api/src/modules/paraclinical-result/paraclinical-result-http.spec.ts, ngay TRƯỚC dòng

// (dùng chung môi trường dựng sẵn của spec đó: prepareEncounterInConsultation, order, payAll, start, rowsOf, resultUrl, baseUrl, fixture, privileged...).
// Rồi chạy: pnpm --filter @nexamed/api exec vitest run src/modules/paraclinical-result/paraclinical-result-http.spec.ts
// Giả định cần kiểm khi chạy: tên trường privileged.paraclinicalResult/paraclinicalResultValue/paraclinicalResultImage/specimenTube/auditLog (cột deletedAt, resultId) và nhãn audit
// 'paraclinical_result.approved', 'specimen_tube.collected'. Ca cuối ("KHOẢNG TRỐNG ĐÃ BIẾT") khẳng định HIỆN TRẠNG (thêm ảnh hai lần ra hai ảnh) — sửa khi có khoá chống ghi trùng.
// ------------------------------------------------------------------------------------------------------------------------------------------------------------------
  /**
   * CHỐNG GHI TRÙNG khi mạng đứt giữa chừng (docs/DECISIONS.md #256): máy chủ đã ghi xong nhưng phản hồi không tới nơi nên người dùng BẤM LẠI (gửi cùng yêu cầu lần
   * hai, liên tiếp) hoặc bấm nhanh hai lần (hai yêu cầu ĐỒNG THỜI). Với khâu nhập kết quả xét nghiệm / CĐHA: không được có hai bản kết quả, hai lần ký, hai bản đính
   * chính, hai ống mẫu cùng lúc được xác nhận lấy.
   */
  describe('chống ghi trùng khi bấm lại / gửi đồng thời (khâu nhập kết quả, #256)', () => {
    async function readyLab() {
      const { encounterId } = await prepareEncounterInConsultation();
      const placed = await order(encounterId, [glucoseId]);
      await payAll(encounterId);
      const itemId = placed.items[0]!.id;
      expect((await start(nurseToken, [itemId])).status).toBe(200);
      const body = (value: string, submit: boolean) => ({ sections: [{ itemId, values: [{ indicatorId: glucoseIndicatorId, valueText: value }] }], submit });
      return { encounterId, itemId, body };
    }
    const activeResults = (itemId: string) => privileged.paraclinicalResult.findMany({ where: { tenantId: fixture.tenantA.id, clinicalOrderItemId: itemId, deletedAt: null } });
    const auditCount = (action: string, entityIds: string[]) => privileged.auditLog.count({ where: { tenantId: fixture.tenantA.id, action, entityId: { in: entityIds } } });

    it('LƯU NHÁP cùng nội dung hai lần liên tiếp và hai lần đồng thời → vẫn đúng 1 bản kết quả, mỗi chỉ số đúng 1 dòng giá trị', async () => {
      const { itemId, body } = await readyLab();
      const first = await http().put(resultUrl(itemId)).set(authed(nurseToken)).send(body('6,1', false));
      const second = await http().put(resultUrl(itemId)).set(authed(nurseToken)).send(body('6,1', false));
      expect([first.status, second.status], JSON.stringify([first.body, second.body])).toEqual([200, 200]);
      const [a, b] = await Promise.all([
        http().put(resultUrl(itemId)).set(authed(nurseToken)).send(body('6,1', false)),
        http().put(resultUrl(itemId)).set(authed(nurseToken)).send(body('6,1', false)),
      ]);
      expect([a.status, b.status], JSON.stringify([a.body, b.body])).toEqual([200, 200]);

      const results = await activeResults(itemId);
      expect(results).toHaveLength(1);
      const values = await privileged.paraclinicalResultValue.findMany({ where: { tenantId: fixture.tenantA.id, resultId: results[0]!.id, deletedAt: null } });
      expect(values).toHaveLength(1);
      expect(values[0]!.valueText).toBe('6,1');
    });

    it('GỬI DUYỆT hai lần liên tiếp → vẫn 1 bản, chưa ký; DUYỆT hai lần liên tiếp → lần 2 bị từ chối, chỉ 1 bản đã ký, chữ ký không đổi', async () => {
      const { itemId, body } = await readyLab();
      expect((await http().put(resultUrl(itemId)).set(authed(nurseToken)).send(body('6,1', true))).status).toBe(200);
      expect((await http().put(resultUrl(itemId)).set(authed(nurseToken)).send(body('6,1', true))).status).toBe(200);
      expect(await activeResults(itemId)).toHaveLength(1);

      const approve1 = await http().post(`${resultUrl(itemId)}/approve`).set(authed(doctorToken)).send(body('6,1', true));
      expect(approve1.status, JSON.stringify(approve1.body)).toBe(200);
      const [signed] = await activeResults(itemId);
      expect(signed!.signedAt).not.toBeNull();

      const approve2 = await http().post(`${resultUrl(itemId)}/approve`).set(authed(doctorToken)).send(body('6,1', true));
      expect(approve2.status).toBeGreaterThanOrEqual(400);
      expect(approve2.status).toBeLessThan(500);
      const after = await activeResults(itemId);
      expect(after).toHaveLength(1);
      expect(after[0]!.signedAt!.getTime()).toBe(signed!.signedAt!.getTime());
      expect(await auditCount('paraclinical_result.approved', [signed!.id])).toBe(1);
    });

    it('DUYỆT bấm nhanh hai lần ĐỒNG THỜI → đúng MỘT lần thành công, MỘT chữ ký, MỘT dòng nhật ký "đã duyệt"', async () => {
      const { itemId, body } = await readyLab();
      expect((await http().put(resultUrl(itemId)).set(authed(nurseToken)).send(body('6,1', true))).status).toBe(200);
      const [a, b] = await Promise.all([
        http().post(`${resultUrl(itemId)}/approve`).set(authed(doctorToken)).send(body('6,1', true)),
        http().post(`${resultUrl(itemId)}/approve`).set(authed(doctorToken)).send(body('6,1', true)),
      ]);
      const statuses = [a.status, b.status].sort();
      expect(statuses[0], JSON.stringify([a.body, b.body])).toBe(200);
      expect(statuses[1]).toBeGreaterThanOrEqual(400);
      expect(statuses[1]).toBeLessThan(500);
      const results = await activeResults(itemId);
      expect(results).toHaveLength(1);
      expect(results[0]!.signedAt).not.toBeNull();
      expect(await auditCount('paraclinical_result.approved', [results[0]!.id])).toBe(1);
    });

    it('ĐÍNH CHÍNH hai lần liên tiếp và hai lần đồng thời → chỉ MỘT bản nháp đính chính; bản đã ký cũ chỉ bị thay MỘT lần', async () => {
      const { itemId, body } = await readyLab();
      expect((await http().put(resultUrl(itemId)).set(authed(nurseToken)).send(body('6,1', true))).status).toBe(200);
      expect((await http().post(`${resultUrl(itemId)}/approve`).set(authed(doctorToken)).send(body('6,1', true))).status).toBe(200);
      const original = (await activeResults(itemId))[0]!;

      const reason = { reason: 'Nhập nhầm chỉ số glucose' };
      const first = await http().post(`${resultUrl(itemId)}/amend`).set(authed(nurseToken)).send(reason);
      expect(first.status, JSON.stringify(first.body)).toBe(200);
      const second = await http().post(`${resultUrl(itemId)}/amend`).set(authed(nurseToken)).send(reason);
      expect(second.status).toBeGreaterThanOrEqual(400);
      expect(second.status).toBeLessThan(500);
      const drafts = (await activeResults(itemId)).filter((r) => r.supersedesId === original.id);
      expect(drafts).toHaveLength(1);

      // Huỷ rồi đính chính lại: hai yêu cầu ĐỒNG THỜI chỉ một cái được.
      expect((await http().post(`${resultUrl(itemId)}/amend/cancel`).set(authed(nurseToken))).status).toBe(200);
      const [c, d] = await Promise.all([
        http().post(`${resultUrl(itemId)}/amend`).set(authed(nurseToken)).send(reason),
        http().post(`${resultUrl(itemId)}/amend`).set(authed(nurseToken)).send(reason),
      ]);
      expect([c.status, d.status].filter((x) => x === 200)).toHaveLength(1);
      expect((await activeResults(itemId)).filter((r) => r.supersedesId !== null)).toHaveLength(1);
    });

    it('XÁC NHẬN LẤY MẪU hai lần liên tiếp và đồng thời → ống chỉ được lấy MỘT lần; MỞ ỐNG nhiều lần không sinh thêm ống', async () => {
      const { encounterId } = await prepareEncounterInConsultation();
      const placed = await order(encounterId, [glucoseId]);
      await payAll(encounterId);
      const row = (await rowsOf(placed.orderNo))[0]!;
      const openUrl = `${baseUrl('lab')}/orders/${row.orderId}/specimen-collection/open`;
      const opened1 = await http().post(openUrl).set(authed(nurseToken)).send();
      const opened2 = await http().post(openUrl).set(authed(nurseToken)).send();
      expect([opened1.status, opened2.status]).toEqual([200, 200]);
      const tubesAfterOpen = await privileged.specimenTube.findMany({ where: { tenantId: fixture.tenantA.id, clinicalOrderId: row.orderId } });
      expect(tubesAfterOpen).toHaveLength(1);

      const tubeId = tubesAfterOpen[0]!.id;
      const collect = () => http().post(`${baseUrl('lab')}/specimen-tubes/collect`).set(authed(nurseToken)).send({ tubes: [{ tubeId, via: 'MANUAL' }] });
      const [a, b] = await Promise.all([collect(), collect()]);
      expect([a.status, b.status].filter((x) => x === 200), JSON.stringify([a.body, b.body])).toHaveLength(1);
      const third = await collect();
      expect(third.status).toBeGreaterThanOrEqual(400);
      expect(third.status).toBeLessThan(500);
      expect(await auditCount('specimen_tube.collected', [tubeId])).toBe(1);
    });

    it('GHI NHẬN IN TEM hai lần: không lỗi, bộ đếm tăng đúng số lần gọi (hành vi đã biết — chỉ lệch số lần in, không hại dữ liệu)', async () => {
      const { encounterId } = await prepareEncounterInConsultation();
      const placed = await order(encounterId, [glucoseId]);
      await payAll(encounterId);
      const row = (await rowsOf(placed.orderNo))[0]!;
      await http().post(`${baseUrl('lab')}/orders/${row.orderId}/specimen-collection/open`).set(authed(nurseToken)).send();
      const tubeId = (await privileged.specimenTube.findFirstOrThrow({ where: { tenantId: fixture.tenantA.id, clinicalOrderId: row.orderId } })).id;
      for (let i = 0; i < 2; i += 1) expect((await http().post(`${baseUrl('lab')}/specimen-tubes/print`).set(authed(nurseToken)).send({ tubeIds: [tubeId] })).status).toBe(200);
      expect((await privileged.specimenTube.findUniqueOrThrow({ where: { id: tubeId } })).printCount).toBe(2);
    });

    it('KHOẢNG TRỐNG ĐÃ BIẾT: THÊM ẢNH CĐHA cùng tệp hai lần → ra HAI ảnh (không chống trùng theo nội dung). Nếu sau này có khoá chống ghi trùng thì sửa ca này', async () => {
      const { encounterId } = await prepareEncounterInConsultation();
      const placed = await order(encounterId, [ultrasoundId]);
      await payAll(encounterId);
      const itemId = placed.items[0]!.id;
      expect((await start(nurseToken, [itemId])).status).toBe(200);
      const png = Buffer.concat([Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]), Buffer.alloc(64, 1)]);
      const upload = () => http().post(`${baseUrl('imaging')}/items/${itemId}/images`).set(authed(nurseToken)).attach('file', png, 'sieu-am.png');
      expect((await upload()).status).toBe(200);
      expect((await upload()).status).toBe(200);
      const result = (await activeResults(itemId))[0]!;
      expect(await privileged.paraclinicalResultImage.count({ where: { tenantId: fixture.tenantA.id, resultId: result.id, deletedAt: null } })).toBe(2);
    });
  });
